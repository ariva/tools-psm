# shell | command line, cursor at its end (a space before the bar = complete the next word) | expectations
# shell: all, or a comma list of bash, zsh, fish. Expectations: +word must
# be offered, -word must not be. zsh and fish offer options only once a
# dash is typed, so option expectations are bash-only or use `--`. Values
# after `=` (`--deep=`, `--json=`) are zsh/fish only: bash splits the word
# at `=` and clap's bash script cannot complete them (not simulated here).
# Blank lines and # are skipped.
all  | psm                      | +procs +report +track +sessions +snapshots +faq
all  | psm re| +report -procs
all  | psm report               | +memory +growth +trend +timeline +meminfo
all  | psm report trend         | -trend -growth -memory
bash | psm report trend         | +--top +--group
all  | psm report trend --| +--top +--group +--metric
all  | psm report trend growth  | -trend -growth
all  | psm procs                | +list +show
all  | psm procs chrome         | -list -show
all  | psm procs chrome --| +--group +--sort
all  | psm procs --group        | +app +cgroup +name +exe +pid
all  | psm procs --sort         | +cpu +mem +threads +count
all  | psm diff --metric        | +total +anon +pss
all  | psm info --by            | +cpu +mem +threads +swap +io
all  | psm sessions             | +activate +compare +export +import +purge +reset
all  | psm sessions export --format | +json +csv
all  | psm snapshots            | +delete +rename +reset +list
all  | psm completions          | +bash +zsh +fish
all  | psm completions bash     | -bash -zsh
all  | psm track --group        | +app +name +pid
zsh,fish | psm --json=| +full +safe +none
bash | psm -p                   | +list +show
bash | psm -p chrome            | -list -show
